create table public.twitter_campaign_tweets (
  id uuid not null default gen_random_uuid (),
  contest_id uuid not null,
  creator_id uuid not null,
  tweet_id text not null,
  tweet_url text not null,
  twitter_username text not null,
  tweet_text text not null,
  tweet_created_at timestamp with time zone not null,
  tweet_media_urls jsonb null default '[]'::jsonb,
  tweet_type text null default 'tweet'::text,
  is_eligible boolean null default false,
  eligibility_reason text null,
  filter_status text null default 'pending'::text,
  likes integer null default 0,
  replies integer null default 0,
  retweets integer null default 0,
  quote_reposts integer null default 0,
  impressions integer null default 0,
  points integer null default 0,
  points_calculated_at timestamp with time zone null,
  first_fetched_at timestamp with time zone not null default now(),
  last_updated_at timestamp with time zone not null default now(),
  last_metrics_fetch_at timestamp with time zone null,
  is_deleted boolean null default false,
  deleted_at timestamp with time zone null,
  deletion_detected_at timestamp with time zone null,
  target_tweet_id text null,
  moderation_status text null default 'pending'::text,
  manual_points_adjustment integer null default 0,
  manual_points_reason text null,
  constraint twitter_campaign_tweets_pkey primary key (id),
  constraint twitter_campaign_tweets_contest_id_tweet_id_key unique (contest_id, tweet_id),
  constraint twitter_campaign_tweets_creator_id_fkey foreign KEY (creator_id) references users (id) on delete CASCADE,
  constraint twitter_campaign_tweets_contest_id_fkey foreign KEY (contest_id) references contests (id) on delete CASCADE,
  constraint twitter_campaign_tweets_filter_status_check check (
    (
      filter_status = any (
        array[
          'pending'::text,
          'eligible'::text,
          'filtered_out'::text,
          'deleted'::text
        ]
      )
    )
  ),
  constraint twitter_campaign_tweets_moderation_status_check check (
    (
      moderation_status = any (
        array[
          'pending'::text,
          'verified'::text,
          'rejected'::text
        ]
      )
    )
  ),
  constraint twitter_campaign_tweets_tweet_type_check check (
    (
      tweet_type = any (
        array[
          'tweet'::text,
          'quote'::text,
          'retweet'::text,
          'reply'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_twitter_tweets_contest on public.twitter_campaign_tweets using btree (contest_id) TABLESPACE pg_default;

create index IF not exists idx_twitter_tweets_creator on public.twitter_campaign_tweets using btree (creator_id) TABLESPACE pg_default;

create index IF not exists idx_twitter_tweets_eligible on public.twitter_campaign_tweets using btree (contest_id, is_eligible, filter_status) TABLESPACE pg_default
where
  (
    (is_eligible = true)
    and (filter_status = 'eligible'::text)
  );

create index IF not exists idx_twitter_tweets_points on public.twitter_campaign_tweets using btree (contest_id, creator_id, points desc) TABLESPACE pg_default;

create index IF not exists idx_twitter_tweets_updated on public.twitter_campaign_tweets using btree (contest_id, last_updated_at desc) TABLESPACE pg_default;

create index IF not exists idx_twitter_tweets_deleted on public.twitter_campaign_tweets using btree (contest_id, is_deleted) TABLESPACE pg_default
where
  (is_deleted = false);

create index IF not exists idx_twitter_tweets_tweet_id on public.twitter_campaign_tweets using btree (tweet_id) TABLESPACE pg_default;

create index IF not exists idx_twitter_tweets_created_at on public.twitter_campaign_tweets using btree (tweet_created_at desc) TABLESPACE pg_default;

create index IF not exists idx_twitter_tweets_target on public.twitter_campaign_tweets using btree (contest_id, target_tweet_id) TABLESPACE pg_default
where
  (target_tweet_id is not null);

create index IF not exists idx_twitter_tweets_moderation_status on public.twitter_campaign_tweets using btree (contest_id, moderation_status) TABLESPACE pg_default
where
  (moderation_status is not null);

create trigger increment_participant_tweet_count_trigger
after INSERT on twitter_campaign_tweets for EACH row
execute FUNCTION increment_participant_tweet_count ();

create trigger update_twitter_tweets_updated_at BEFORE
update on twitter_campaign_tweets for EACH row
execute FUNCTION update_twitter_tweet_updated_at ();
