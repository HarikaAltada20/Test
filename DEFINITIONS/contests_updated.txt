-- Updated contests table definition with new features
-- Date: 2025-10-01

create table public.contests (
  id uuid not null default gen_random_uuid (),
  advertiser_id uuid null,
  title text not null,
  platform text null,
  start_date timestamp with time zone null,
  end_date timestamp with time zone null,
  thumbnail_url text null,
  resources jsonb null,
  category text null,
  inspiration_links jsonb null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  contest_type public.contest_type_enum null,
  contest_based_details jsonb null,
  live_submission_count integer null default 0,
  post_contest_status public.post_contest_status_enum null,
  brief_html text null,
  brief_json jsonb null,
  last_metrics_updated timestamp with time zone null,
  rules_html text null,
  rules_json jsonb null,
  moderation_status public.contest_moderation_status_enum null default 'draft'::contest_moderation_status_enum,
  submitted_for_approval_at timestamp with time zone null,
  approved_at timestamp with time zone null,
  approved_by uuid null,
  published_at timestamp with time zone null,
  rejection_reason text null,
  payment_details jsonb null,
  subscription_info_of_user jsonb null,
  
  -- New features (2025-10-01)
  multiple_submissions_enabled boolean null default false,
  max_submissions_per_creator integer null default 1,
  content_type text null check (content_type in ('ugc', 'clipping', 'other')),
  bonus_details jsonb null,
  max_earnings_per_creator integer null,
  -- Note: flat_fee_bonus is stored in contest_based_details JSONB (optional field, in cents)
  
  constraint contests_pkey primary key (id),
  constraint contests_advertiser_id_fkey foreign key (advertiser_id) references advertiser_profiles (id),
  constraint contests_approved_by_fkey foreign key (approved_by) references auth.users (id)
) tablespace pg_default;

-- Indexes
create index if not exists idx_contests_moderation_status on public.contests using btree (moderation_status) tablespace pg_default;
create index if not exists idx_contests_post_contest_status on public.contests using btree (post_contest_status) tablespace pg_default;
create index if not exists idx_contests_brief_html on public.contests using gin (to_tsvector('english'::regconfig, brief_html)) tablespace pg_default;
create index if not exists idx_contests_last_metrics_updated on public.contests using btree (last_metrics_updated) tablespace pg_default;
create index if not exists idx_contests_rules_html on public.contests using gin (to_tsvector('english'::regconfig, rules_html)) tablespace pg_default;
create index if not exists idx_contests_approved_by on public.contests using btree (approved_by) tablespace pg_default where (approved_by is not null);
create index if not exists idx_contests_subscription_info_of_user on public.contests using gin (subscription_info_of_user) tablespace pg_default;
create index if not exists idx_contests_content_type on public.contests(content_type) where content_type is not null;
create index if not exists idx_contests_multiple_submissions on public.contests(multiple_submissions_enabled) where multiple_submissions_enabled = true;

-- Trigger
create trigger on_contests_update before update on contests for each row execute function handle_updated_at();
