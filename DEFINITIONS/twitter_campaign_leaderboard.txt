create table public.twitter_campaign_leaderboard (
  id uuid not null default gen_random_uuid (),
  contest_id uuid not null,
  creator_id uuid not null,
  total_points integer null default 0,
  total_eligible_tweets integer null default 0,
  total_likes integer null default 0,
  total_replies integer null default 0,
  total_retweets integer null default 0,
  total_quote_reposts integer null default 0,
  total_impressions integer null default 0,
  current_rank integer null,
  last_refreshed_at timestamp with time zone not null default now(),
  next_refresh_available_at timestamp with time zone not null default (now() + '01:00:00'::interval),
  refresh_count integer null default 0,
  manual_points_adjustment integer null default 0,
  manual_points_reason text null,
  moderation_status text null default 'pending'::text,
  rejection_reason text null,
  paid_at timestamp with time zone null,
  earnings integer null default 0,
  paid_rank integer null,
  constraint twitter_campaign_leaderboard_pkey primary key (id),
  constraint twitter_campaign_leaderboard_contest_id_creator_id_key unique (contest_id, creator_id),
  constraint twitter_campaign_leaderboard_contest_id_fkey foreign KEY (contest_id) references contests (id) on delete CASCADE,
  constraint twitter_campaign_leaderboard_creator_id_fkey foreign KEY (creator_id) references users (id) on delete CASCADE,
  constraint twitter_campaign_leaderboard_moderation_status_check check (
    (
      moderation_status = any (
        array[
          'pending'::text,
          'verified'::text,
          'rejected'::text,
          'paid'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_twitter_leaderboard_contest on public.twitter_campaign_leaderboard using btree (contest_id) TABLESPACE pg_default;

create index IF not exists idx_twitter_leaderboard_rank on public.twitter_campaign_leaderboard using btree (contest_id, current_rank) TABLESPACE pg_default;

create index IF not exists idx_twitter_leaderboard_points on public.twitter_campaign_leaderboard using btree (contest_id, total_points desc) TABLESPACE pg_default;

create index IF not exists idx_twitter_leaderboard_refresh on public.twitter_campaign_leaderboard using btree (contest_id, next_refresh_available_at) TABLESPACE pg_default;

create index IF not exists idx_twitter_leaderboard_moderation_status on public.twitter_campaign_leaderboard using btree (contest_id, moderation_status) TABLESPACE pg_default
where
  (moderation_status is not null);

create index IF not exists idx_twitter_leaderboard_moderation_paid on public.twitter_campaign_leaderboard using btree (contest_id) TABLESPACE pg_default
where
  (moderation_status = 'paid'::text);
